\newglossaryentry{C'Space}{
    name={C'Space},
    description={
        Campagne nationale de lancements de fusées expérimentales organisée par le \acrfull{CNES} et
        l'association Planète Sciences. Elle est exculsivement réservée aux étudiants des écoles
        d'ingénieurs et universités françaises. Cette campagne se déroule chaque année dans le camp
        de Ger à Tarbes dans les Hautes-Pyrénées au début du mois de juillet.
    },
}
\newglossaryentry{APEX}{
    name={APEX},
    description={
        Système de mesure embarqué développé par l'association AéroIPSA pour la collecte, l'enregistrement
        et la télémesure des données d'avionique durant le vol de fusée expérimentale. Le lien vers le
        dossier de projet est disponible sur le site de l'AéroIPSA : \href{https://www.aeroipsa.com/apex}{https://www.aeroipsa.com/apex}
        ou directement via ce \href{https://static1.squarespace.com/static/5b641883710699360ad48c74/t/68b4c939bce327051e2adc68/1756678472172/Rapport-Projet-APEX.pdf}{lien}.
    },
}
